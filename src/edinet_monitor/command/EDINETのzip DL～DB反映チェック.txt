import os
import sqlite3
import unicodedata
from pathlib import Path
from datetime import datetime

DB_PATH = r"E:\EDINET_Data\edinet_monitor\db\edinet_monitor.db"
DATE_FROM = os.environ["CHECK_DATE_FROM"]
DATE_TO = os.environ["CHECK_DATE_TO"]
FORM_CODES = [x.strip() for x in os.environ["CHECK_FORM_CODES"].split(",") if x.strip()]
OUT_DIR = Path(os.environ["CHECK_OUT_DIR"])

def display_width(text):
    width = 0
    for ch in str(text):
        width += 2 if unicodedata.east_asian_width(ch) in ("F", "W", "A") else 1
    return width

def pad_right(text, width):
    text = str(text)
    return text + " " * max(0, width - display_width(text))

def pad_left(text, width):
    text = str(text)
    return " " * max(0, width - display_width(text)) + text

conn = sqlite3.connect(f"file:{DB_PATH}?mode=ro", uri=True)
conn.row_factory = sqlite3.Row
cur = conn.cursor()

placeholders = ",".join("?" for _ in FORM_CODES)
rows = cur.execute(
    f"""
    select
      substr(f.submit_date, 1, 10) as submit_date,
      f.form_type,
      count(*) as total,
      sum(case when coalesce(f.download_status, '') = 'downloaded' then 1 else 0 end) as downloaded,
      sum(case when coalesce(f.xbrl_path, '') <> '' then 1 else 0 end) as xbrl_ready,
      sum(case when coalesce(f.parse_status, '') = 'raw_facts_saved' then 1 else 0 end) as raw_only,
      sum(case when coalesce(f.parse_status, '') = 'normalized_metrics_saved' then 1 else 0 end) as normalized_only,
      sum(case when coalesce(f.parse_status, '') = 'derived_metrics_saved' then 1 else 0 end) as derived_saved,
      sum(case when coalesce(f.parse_status, '') like '%error' then 1 else 0 end) as errors
    from filings f
    join issuer_master im
      on im.edinet_code = f.edinet_code
    where f.form_type in ({placeholders})
      and im.is_listed = 1
      and im.exchange = 'TSE'
      and substr(coalesce(f.submit_date, ''), 1, 10) between ? and ?
    group by substr(f.submit_date, 1, 10), f.form_type
    order by submit_date, f.form_type
    """,
    (*FORM_CODES, DATE_FROM, DATE_TO),
).fetchall()

OUT_DIR.mkdir(parents=True, exist_ok=True)
out_path = OUT_DIR / f"submit_date_db_reflection_check_{DATE_FROM}_to_{DATE_TO}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"

cols = [
    ("submit_date", "提出日", "left"),
    ("form_type", "form", "left"),
    ("total", "総数", "right"),
    ("downloaded", "ZIP", "right"),
    ("xbrl_ready", "XBRL", "right"),
    ("raw_only", "RAW", "right"),
    ("normalized_only", "NORMALIZED", "right"),
    ("derived_saved", "DERIVED", "right"),
    ("errors", "ERROR", "right"),
]

widths = {}
for key, label, _ in cols:
    widths[key] = display_width(label) if not rows else max(display_width(label), *(display_width(row[key]) for row in rows))

lines = []
lines.append(f"generated_at: {datetime.now().isoformat(timespec='seconds')}")
lines.append(f"submit_date_from: {DATE_FROM}")
lines.append(f"submit_date_to: {DATE_TO}")
lines.append(f"form_codes: {','.join(FORM_CODES)}")
lines.append("")
lines.append(" | ".join(
    pad_left(label, widths[key]) if align == "right" else pad_right(label, widths[key])
    for key, label, align in cols
))
lines.append("-+-".join("-" * widths[key] for key, _, _ in cols))

for row in rows:
    lines.append(" | ".join(
        pad_left(row[key], widths[key]) if align == "right" else pad_right(row[key], widths[key])
        for key, _, align in cols
    ))

out_path.write_text("\n".join(lines) + "\n", encoding="utf-8-sig")
print(f"saved: {out_path}")

conn.close()
